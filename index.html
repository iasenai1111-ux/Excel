import test from 'node:test'; import assert from 'node:assert/strict'; import { LEVELS } from '../js/phases.js'; import { computeResult, isFormulaAccepted, normalizeFormula } from '../js/miniExcel.js'; test('normalizeFormula removes spaces and uppercase accents', () => { assert.equal(normalizeFormula(' = média ( b2 : d2 ) '), '=MÉDIA(B2:D2)'); }); test('Fase 1 aceita soma correta', () => { const phase = LEVELS[0]; assert.equal(isFormulaAccepted(phase, '=B2+C2'), true); assert.equal(computeResult(phase, '=B2+C2'), 200); }); test('Fase 2 aceita subtração correta', () => { const phase = LEVELS[1]; assert.equal(isFormulaAccepted(phase, '=B2-C2'), true); assert.equal(computeResult(phase, '=B2-C2'), 12); }); test('Fase 3 aceita multiplicação correta', () => { const phase = LEVELS[2]; assert.equal(isFormulaAccepted(phase, '=B2*C2'), true); assert.equal(computeResult(phase, '=B2*C2'), 270); }); test('Fase 4 aceita média correta', () => { const phase = LEVELS[3]; assert.equal(isFormulaAccepted(phase, '=MEDIA(B2:D2)'), true); assert.equal(computeResult(phase, '=MEDIA(B2:D2)'), 8); }); test('Fase 5 aceita desconto de 15%', () => { const phase = LEVELS[4]; assert.equal(isFormulaAccepted(phase, '=B2*(1-C2)'), true); assert.equal(computeResult(phase, '=B2*(1-C2)'), 170); }); test('fórmula incorreta é rejeitada', () => { const phase = LEVELS[0]; assert.equal(isFormulaAccepted(phase, '=B2-C2'), false); });
